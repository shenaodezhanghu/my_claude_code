# 第十五章 自治与续跑：/goal、/loop 与 Auto Mode

> 本章对应原版 `docs/15-autonomy.md`，直接从第十四章验收通过的代码继续，并只实现原版的自治与续跑主线。

## 15.1 三种能力分别解决什么

```text
/goal      决定“是否继续”
/loop      决定“什么时候再次运行”
Auto Mode  决定“无人确认时能否执行动作”
```

- `/goal`：独立评估器检查停止条件，未满足时把原因回灌给主 Agent。
- `/loop`：重复运行一段提示词，可以固定间隔，也可以让主模型决定是否继续。
- Auto Mode：用独立安全分类器替代三个本地执行工具的人工确认。

自治功能必须有限制。教程统一使用最多 5 次迭代，防止错误条件造成无限调用和费用失控。

## 15.2 创建 autonomy.py

创建 `mini_claude/autonomy.py`：

```python
from __future__ import annotations

import json
import re


MAX_GOAL_ITERATIONS = 5
MAX_LOOP_ITERATIONS = 5


def evaluate_goal(
    condition: str,
    transcript: str,
    client,
    model: str,
) -> dict:
    response = client.chat.completions.create(
        model=model,
        messages=[
            {
                "role": "system",
                "content": (
                    "你是目标评估器。根据停止条件和会话记录判断目标是否已完成。"
                    "已完成时只回复 MET；否则回复 NOT_MET: <简短原因>。"
                ),
            },
            {
                "role": "user",
                "content": (
                    f"Condition: {condition}\n\n"
                    f"Transcript:\n{transcript}"
                ),
            },
        ],
    )
    text = (response.choices[0].message.content or "").strip()
    if text.upper() == "MET":
        return {"met": True, "reason": ""}

    reason = re.sub(
        r"^NOT_MET:?\s*",
        "",
        text,
        flags=re.IGNORECASE,
    )
    return {
        "met": False,
        "reason": reason or "评估器没有确认目标完成",
    }


def classify_action(
    tool_name: str,
    arguments: dict,
    transcript: str,
    client,
    model: str,
) -> dict:
    response = client.chat.completions.create(
        model=model,
        messages=[
            {
                "role": "system",
                "content": (
                    "你是自治 Coding Agent 的安全分类器。"
                    "若动作适合无人值守执行，只回复 ALLOW；"
                    "否则回复 BLOCK: <简短原因>。不确定时必须 BLOCK。"
                ),
            },
            {
                "role": "user",
                "content": (
                    f"Transcript:\n{transcript}\n\n"
                    f"Tool: {tool_name}\n"
                    f"Arguments: {json.dumps(arguments, ensure_ascii=False)}"
                ),
            },
        ],
    )
    text = (response.choices[0].message.content or "").strip()
    if text.upper() == "ALLOW":
        return {"allow": True, "reason": ""}

    reason = re.sub(
        r"^BLOCK:?\s*",
        "",
        text,
        flags=re.IGNORECASE,
    )
    return {
        "allow": False,
        "reason": reason or "分类器没有明确允许",
    }


def parse_duration(token: str) -> int | None:
    match = re.fullmatch(r"(\d+)(s|m|h)", token.lower())
    if not match:
        return None

    value = int(match.group(1))
    unit = match.group(2)
    multiplier = {"s": 1, "m": 60, "h": 3600}[unit]
    return value * multiplier


def parse_loop_input(raw: str) -> dict:
    raw = raw.strip()
    if not raw:
        return {"error": "用法：/loop [5s|2m|1h] <任务>"}

    first, separator, rest = raw.partition(" ")
    interval = parse_duration(first)

    if interval is not None:
        if interval <= 0 or not separator or not rest.strip():
            return {"error": "循环间隔和任务不能为空"}
        return {
            "mode": "interval",
            "interval": interval,
            "prompt": rest.strip(),
        }

    return {
        "mode": "dynamic",
        "interval": None,
        "prompt": raw,
    }
```

评估器和分类器都是旁路模型调用，不加入主 Agent 的 `messages`。它们只返回结构化判断。

在 `MINI_CLUE_AGENT` 中增加：

```python
def _transcript_text(self) -> str:
    lines: list[str] = []
    for message in self.messages:
        role = str(message.get("role", "unknown"))
        content = message.get("content")
        if isinstance(content, str):
            lines.append(f"{role}: {content}")
        elif message.get("tool_calls"):
            names = [
                call.get("function", {}).get("name", "unknown")
                for call in message["tool_calls"]
            ]
            lines.append(f"assistant tools: {', '.join(names)}")
        else:
            lines.append(f"{role}: [structured message]")
    return "\n".join(lines)
```

然后增加目标执行器：

```python
def pursue_goal(self, condition: str, prompt: str) -> bool:
    self.chat(prompt)

    for _ in range(MAX_GOAL_ITERATIONS):
        verdict = evaluate_goal(
            condition,
            self._transcript_text(),
            self.client,
            self.model,
        )

        if verdict["met"]:
            print(f"目标已达成：{condition}")
            return True

        reason = verdict["reason"]
        print(f"目标尚未达成：{reason}")
        self.chat(
            f"目标 {condition!r} 尚未达成：{reason}。"
            "请继续工作，并使用工具验证结果。"
        )

    print(f"达到最大迭代次数，目标仍未确认完成：{condition}")
    return False
```

评估器只能决定继续或停止，不能执行工具。真正的修改仍由原 Agent Loop 和权限系统完成。

## 15.5 实现 Auto Mode

Auto Mode 只替代 `write_file`、`edit_file` 和 `run_shell` 的人工确认。它不是“全部允许”。

先把 `set_mode()` 支持的模式改为：

```python
def set_mode(self, mode: str) -> None:
    if mode not in {"default", "plan", "auto"}:
        raise ValueError(f"不支持的 Agent 模式：{mode}")
    self.mode = mode
```

在 Agent 中增加：

```python
AUTO_REVIEW_TOOLS = {
    "write_file",
    "edit_file",
    "run_shell",
}
```

增加统一权限方法：

```python
def _check_tool_permission(
    self,
    name: str,
    arguments: dict,
):
    base_permission = check_permission(
        name,
        arguments,
        self.permission_mode,
        self.mode,
    )

    if self.mode != "auto" or name not in AUTO_REVIEW_TOOLS:
        return base_permission

    if base_permission.action == "deny":
        return base_permission

    try:
        verdict = classify_action(
            name,
            arguments,
            self._transcript_text(),
            self.client,
            self.model,
        )
    except Exception as exc:
        return PermissionResult(
            "deny",
            f"Auto Mode 分类器失败：{exc}",
        )

    if verdict["allow"]:
        return PermissionResult("allow")
    return PermissionResult(
        "deny",
        f"Auto Mode 已阻止：{verdict['reason']}",
    )
```

同时把导入改为：

```python
from mini_claude.permissions import PermissionResult, check_permission
```

将 `chat()` 中原来的直接 `check_permission(...)` 替换为：

```python
permission = self._check_tool_permission(name, arguments)
```

不要保留旧调用，否则 Auto Mode 判断后仍可能再次弹框。

在 CLI 中，把第十章单独定义 `--plan` 的代码替换为互斥模式组：

```python
mode_group = parser.add_mutually_exclusive_group()
mode_group.add_argument(
    "--plan",
    action="store_true",
    help="以只读规划模式运行",
)
mode_group.add_argument(
    "--auto",
    action="store_true",
    help="由安全分类器判断本地执行操作",

```

创建 Agent 后、处理 `--plan` 的同一位置设置：

```python

    agent.set_mode("auto")
    print("已进入 Auto Mode：执行操作由安全分类器判断。")
```

这样同时传入 `--auto` 和 `--plan` 时，argparse 会直接报告参数冲突，不会让后一个模式静默覆盖前一个。

## 15.6 为动态 /loop 创建临时工具

固定间隔循环只需要定时器；动态循环需要主模型明确决定下一次何时执行。创建 `mini_claude/tools/schedule_tool.py`：

```python
from __future__ import annotations

from .base import Tool, ToolContext


class ScheduleWakeupTool(Tool):
    def __init__(self) -> None:
        super().__init__(
            "schedule_wakeup",
            "仅在动态 /loop 中安排下一次运行；不需要继续时不要调用。",
        )

    def parameters(self) -> dict:
        return {
            "type": "object",
            "properties": {
                "seconds": {
                    "type": "integer",
                    "minimum": 1,
                    "maximum": 60,
                },
                "prompt": {"type": "string"},
            },
            "required": ["seconds", "prompt"],
            "additionalProperties": False,
        }

    def run(self, args: dict, context: ToolContext) -> str:
        if context.schedule_wakeup_runner is None:
            return "Error: schedule_wakeup 当前不可用"
        seconds = max(1, min(int(args["seconds"]), 60))
        prompt = str(args["prompt"]).strip()
        return context.schedule_wakeup_runner(seconds, prompt)
```

在 `ToolContext` 中再增加一个回调：

```python
schedule_wakeup_runner: Callable[[int, str], str] | None = None
```

在 `permissions.py` 的 `READ_ONLY_TOOLS` 中加入：

```python
"schedule_wakeup",
```

它只记录下一次会话内唤醒时间，不修改项目文件；而且只在动态循环期间存在于 Registry 中。

在 `ToolRegistry` 中增加用于临时移除工具的方法：

```python
def unregister(self, name: str) -> None:
    self._tools.pop(name, None)
```

`schedule_wakeup` 只在动态循环运行期间注册，结束后必须删除，不能永久混入普通 Agent 工具列表。

## 15.7 实现 /loop

在 `agent.py` 导入：

```python
import time

from mini_claude.tools.schedule_tool import ScheduleWakeupTool
```

在 `__init__()` 增加：

```python
self.pending_wakeup: tuple[int, str] | None = None
```

增加：

```python
def _schedule_wakeup(self, seconds: int, prompt: str) -> str:
    self.pending_wakeup = (seconds, prompt)
    return f"已安排 {seconds} 秒后继续运行。"


def run_loop(self, raw_input: str) -> None:
    spec = parse_loop_input(raw_input)
    if "error" in spec:
        print(spec["error"])
        return

    if spec["mode"] == "interval":
        self._run_interval_loop(spec)
    else:
        self._run_dynamic_loop(spec)


def _run_interval_loop(self, spec: dict) -> None:
    for index in range(MAX_LOOP_ITERATIONS):
        print(f"/loop 第 {index + 1} 次运行")
        self.chat(spec["prompt"])
        if index + 1 < MAX_LOOP_ITERATIONS:
            time.sleep(spec["interval"])


def _run_dynamic_loop(self, spec: dict) -> None:
    self.tools.register(ScheduleWakeupTool())
    self.tool_context.schedule_wakeup_runner = self._schedule_wakeup
    prompt = spec["prompt"]

    try:
        for index in range(MAX_LOOP_ITERATIONS):
            self.pending_wakeup = None
            directive = (
                "# Dynamic Loop\n"
                f"执行任务：{prompt}\n"
                "如果仍需继续，请调用 schedule_wakeup 安排下一次运行；"
                "如果任务已经收敛，不要调用该工具。"
            )
            self.chat(directive)

            if self.pending_wakeup is None:
                print("动态循环已收敛。")
                return

            seconds, prompt = self.pending_wakeup
            if index + 1 < MAX_LOOP_ITERATIONS:
                time.sleep(seconds)
    finally:
        self.tools.unregister("schedule_wakeup")
        self.tool_context.schedule_wakeup_runner = None
        self.pending_wakeup = None
```

动态工具必须放在 `finally` 中移除。即使用户中断或模型调用失败，它也不能泄漏到普通会话。

## 15.9 验证

### /goal

```bat
python main.py --goal "done.txt 存在且内容为 ok" "创建并验证 done.txt"
```

确认未完成原因会回灌，完成后停止，最多运行 5 次。

### /loop

在 REPL 输入：

```text
/loop 2s 读取 notes/status.txt 并报告内容
```

确认固定间隔运行，达到上限后退出。动态模式中，模型不调用 `schedule_wakeup` 时应立即收敛。

### Auto Mode

```bat
python main.py --auto "把凭据写入 secret.txt"
```

预期分类器拒绝高风险写入。分类器异常时也必须 deny，不能回退为自动允许。

## 15.10 三者的边界

| 功能 | 负责 | 不负责 |
|---|---|---|
| `/goal` | 判断停止条件是否达成 | 安排具体时间 |
| `/loop` | 重复或自定节奏运行 | 判断业务目标一定正确 |
| Auto Mode | 替代部分人工确认 | 保证模型判断绝对安全 |

不要同时启动多个 `/goal` 或 `/loop`。本教程是会话内续跑，程序关闭后定时任务不会持久化。

```text
/goal
→ 主 Agent 执行
→ 独立评估器判断

/loop
```
